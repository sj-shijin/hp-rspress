#include <concepts>
#include <iostream>
#include <span>

#if defined(USE_FLOAT)
using Scalar = float;
#else
using Scalar = double;
#endif

// 只读视图不复制数据；点积要求向量等长。
template <std::floating_point T>
class Vec {
public:
    explicit Vec(std::span<const T> data) : data_(data) {}
    [[nodiscard]] T dot(const Vec& rhs) const noexcept {
        T total{};
        for (std::size_t i = 0; i < data_.size(); ++i)
            total += data_[i] * rhs.data_[i];
        return total;
    }
private:
    std::span<const T> data_;
};

int main() {
    constexpr Scalar x[]{1.0, 2.0, 3.0}, y[]{4.0, 5.0, 6.0};
    const Vec<Scalar> a{x}, b{y};
    // 引用捕获向量，初始化捕获权重。
    const auto score = [&a, scale = 0.5](const auto& v) {
        return scale * a.dot(v);
    };
    std::cout << "加权点积 = " << score(b) << '\n';
}

from dataclasses import dataclass
from math import fsum


@dataclass(frozen=True)
class Vector:
    values: tuple[float, ...]

    def dot(self, other: "Vector") -> float:
        """计算等长向量的点积，长度不符时抛出异常。"""
        return fsum(
            x * y for x, y in zip(self.values, other.values, strict=True)
        )


def describe(name: str, value: float | None) -> str:
    # 同时观察参数、类型、常量、条件表达式和格式化字符串。
    return f"{name}: {value:.3f}" if value is not None else "暂无结果"


if __name__ == "__main__":
    lhs = Vector((1.0, 2.0, 3.0))
    rhs = Vector((4.0, 5.0, 6.0))
    print(describe("点积", lhs.dot(rhs)))

      {pending.map(task => (

  );

<Tab value="sql" label={languageLabel("sql")}>

```sql title="tasks.sql · SQL" lineNumbers
WITH tasks(project, title, done) AS (
    VALUES
        ('Cinderfolio', '调整配色', TRUE),
        ('Cinderfolio', '导出主题', FALSE),
        ('Pele', '检查图表', FALSE)
)
SELECT project, COUNT(*) AS pending
FROM tasks
WHERE done = FALSE
GROUP BY project
HAVING COUNT(*) > 0
ORDER BY pending DESC;
```
