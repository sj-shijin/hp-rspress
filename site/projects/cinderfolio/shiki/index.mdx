```c title="tasks.c · C11" lineNumbers
#include <stdbool.h>
#include <stdio.h>
#define MAX_SHOWN 2

typedef struct { const char *title; bool done; } Task;
typedef bool (*Predicate)(const Task *);

static bool is_pending(const Task *task) {
    return !task->done;
}

// 回调决定筛选条件，指针访问原数据，达到上限后提前结束。
static size_t visit(const Task tasks[], size_t count, Predicate accept) {
    size_t shown = 0;
    for (size_t i = 0; i < count; ++i) {
        const Task *task = &tasks[i];
        if (!accept(task)) continue;
        printf("%zu. %s\n", shown + 1, task->title);
        if (++shown == MAX_SHOWN) break;
    }
    return shown;
}

int main(void) {
    const Task tasks[] = {
        {.title = "调整配色", .done = true},
        {.title = "导出主题", .done = false},
        {.title = "检查示例", .done = false},
    };
#if defined(DEBUG)
    fprintf(stderr, "开始筛选待办任务\n");
#endif
    size_t count = sizeof tasks / sizeof tasks[0];
    size_t shown = visit(tasks, count, is_pending);
    return shown > 0 ? 0 : 1;
}
```

```cpp title="vector.cpp · C++20" lineNumbers
template <std::floating_point T>

public:
    explicit Vec(std::span<const T> data) : data_(data) {}
    [[nodiscard]] T dot(const Vec& rhs) const {
        if (data_.size() != rhs.data_.size())
            throw std::invalid_argument("向量长度不一致");
        T total{};
        for (std::size_t i = 0; i < data_.size(); ++i)
```

```python title="tasks.py · Python 3.10+" lineNumbers
import json
from dataclasses import dataclass
from pathlib import Path

@dataclass(frozen=True)
class Task:
    id: int
    title: str
    done: bool = False
    owner: str | None = None

def load_tasks(path: Path) -> list[Task]:
    """读取任务；文件关闭由上下文管理器负责。"""
    with path.open(encoding="utf-8") as stream:
        data = json.load(stream)
    return [Task(**item) for item in data["tasks"]]

def summarize(tasks: list[Task]) -> dict[str, int]:
    counts: dict[str, int] = {}
    for task in tasks:
        if task.done:
            continue
        owner = task.owner or "未分配"
        counts[owner] = counts.get(owner, 0) + 1
    return counts

if __name__ == "__main__":
    try:
        tasks = load_tasks(Path("tasks.json"))
        for owner, count in summarize(tasks).items():
            print(f"{owner}: {count:02d} 项待办")
    except (OSError, ValueError) as error:
        print(f"读取失败：{error}")
```

```javascript title="tasks.js · JavaScript" lineNumbers
async function loadTasks(url, { limit = 3 } = {}) {
    const response = await fetch(url);
    if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
    }
    const { tasks = [] } = await response.json();
    // 先筛选，再排序；返回适合展示的数据。
    return tasks
        .filter(task => !task.done)
        .sort((a, b) => a.title.localeCompare(b.title))
        .slice(0, limit)
        .map(({ id, title, owner }) => ({
            id, title: title.trim(), owner: owner ?? "未分配",
        }));
}

async function main() {
    try {
        const pending = await loadTasks("/tasks.json", {
            limit: 5,
        });
        if (pending.length === 0) {
            console.log("全部完成");
            return;
        }
        for (const { title, owner } of pending) {
            console.log(`${owner}: ${title}`);
        }
    } catch (error) {
        console.error("读取失败：", error.message);
    }
}
```

```typescript title="tasks.ts · TypeScript 4.9+" lineNumbers
interface Task {
    readonly id: number;
    title: string;
    done: boolean;
    owner?: string | null;
}
type State<T> =
    | { status: "ready"; data: T }
    | { status: "error"; message: string };

function pending<T extends Task>(items: readonly T[]): T[] {
    return items.filter(({ done }) => !done);
}

function summarize(state: State<readonly Task[]>): string {
    // 判别字段决定后续流程，也缩小可访问的类型。
    switch (state.status) {
        case "error":
            return `读取失败：${state.message}`;
        case "ready": {
            const names = pending(state.data).map(task => {
                const owner = task.owner?.trim() || "未分配";
                return `${owner}: ${task.title}`;
            });
            return names.join("\n") || "全部完成";
        }
    }
}

const state = {
    status: "ready",
    data: [{
        id: 1, title: "导出主题", done: false, owner: null,
    }],
} satisfies State<Task[]>;
console.log(summarize(state));
```

```csharp title="Program.cs · C# 10 / .NET 6+" lineNumbers
record TaskItem(int Id, string Title, bool Done);

class Program {
    static async Task<TaskItem[]> LoadAsync() {
        await Task.Yield();
        return new[] {
            new TaskItem(1, "调整配色", true),
            new TaskItem(2, "导出主题", false),
        };
    }

    static string Describe(TaskItem task) => task switch {
        { Done: true } => "已完成",
        { Title.Length: 0 } => throw new ArgumentException("标题为空"),
        _ => $"{task.Id}: {task.Title}",
    };

    static async Task Main() {
        try {
            var tasks = await LoadAsync();
            // LINQ 按顺序筛选和排序，模式匹配决定输出内容。
            var pending = tasks.Where(task => !task.Done)
                .OrderBy(task => task.Id).ToArray();
            if (pending.Length == 0) {
                Console.WriteLine("全部完成");
                return;
            }
            foreach (var task in pending)
                Console.WriteLine(Describe(task));
        } catch (ArgumentException error) {
            Console.Error.WriteLine(error.Message);
        }
    }
}
```

```rust title="tasks.rs · Rust 2021" lineNumbers
#[derive(Debug)]
struct Task<'a> {
    title: &'a str,
    done: bool,
}

fn parse_task(line: &str) -> Result<Task<'_>, String> {
    let (title, state) = line
        .split_once(',')
        .ok_or("缺少状态")?;
    let done = match state.trim() {
        "done" => true,
        "todo" => false,
        other => return Err(format!("未知状态：{other}")),
    };
    Ok(Task { title: title.trim(), done })
}

fn main() -> Result<(), String> {
    // 任意一行失败就返回错误；成功时保留借用的标题。
    let tasks = ["调整配色,done", "导出主题,todo"]
        .into_iter()
        .map(parse_task)
        .collect::<Result<Vec<_>, _>>()?;
    let pending: Vec<_> = tasks.iter()
        .filter(|task| !task.done)
        .map(|task| task.title)
        .collect();

    if let Some(first) = pending.first() {
        println!("下一项：{first}");
    } else {
        println!("全部完成");
    }
    println!("{} 项：{}", pending.len(), pending.join(", "));
    Ok(())
}
```

```go title="tasks.go · Go" lineNumbers
    "encoding/json"

    "os"

type Task struct {
    Title string `json:"title"`
    Done  bool   `json:"done"`
}

func loadTasks(path string) ([]Task, error) {
    file, err := os.Open(path)
    if err != nil {
        return nil, err
    }
    defer file.Close()
    var data struct { Tasks []Task `json:"tasks"` }
    if err := json.NewDecoder(file).Decode(&data); err != nil {
        return nil, fmt.Errorf("解析任务：%w", err)
    }
    return data.Tasks, nil
}

func main() {
    tasks, err := loadTasks("tasks.json")
    if err != nil {
        fmt.Fprintln(os.Stderr, err)
        return
    }
    // 已完成的任务直接跳过；只输出待办。
    for index, task := range tasks {
        if task.Done { continue }
        fmt.Printf("%d. %s\n", index + 1, task.Title)
    }
}
```

```java title="Tasks.java · Java 17+" lineNumbers
import java.util.List;

public class Tasks {
    enum State { TODO, DONE }
    record Task(String title, State state) {
        Task {
            if (title.isBlank()) {
                throw new IllegalArgumentException("标题为空");
            }
        }
    }

    static String label(Task task) {
        return switch (task.state()) {
            case TODO -> "待办：" + task.title();
            case DONE -> "完成：" + task.title();
        };
    }

    public static void main(String[] args) {
        try {
            var tasks = List.of(
                new Task("调整配色", State.DONE),
                new Task("导出主题", State.TODO)
            );
            // 流操作按顺序表达筛选和转换。
            var pending = tasks.stream()
                .filter(task -> task.state() == State.TODO)
                .map(Tasks::label)
                .toList();
            for (var line : pending) {
                System.out.println(line);
            }
        } catch (IllegalArgumentException error) {
            System.err.println(error.getMessage());
        }
    }
}
```

```jsx title="TaskList.jsx · React JSX" lineNumbers
import { useState } from "react";

export default function TaskList({ initialTasks = [] }) {
  const [tasks, setTasks] = useState(initialTasks);
  const pending = tasks.filter(task => !task.done);

  function toggle(id) {
    // 返回新对象，让状态更新与事件的关系更清楚。
    setTasks(current => current.map(task =>
      task.id === id ? { ...task, done: !task.done } : task
    ));
  }

  return (
    <section aria-label="待办任务">
      <h2>待办 {pending.length} 项</h2>
      {tasks.length === 0 ? <p>暂无任务</p> : (
        <ul>
          {tasks.map(({ id, title, done }) => (
            <li key={id}>
              <button
                aria-pressed={done}
                onClick={() => toggle(id)}
              >
                {done ? "✓ " : "○ "}{title}
              </button>
            </li>
          ))}
        </ul>
      )}
    </section>
  );
}
```

```sql title="tasks.sql · SQL" lineNumbers
WITH projects(id, name) AS (
    VALUES (1, 'Cinderfolio'), (2, 'Pele')
), tasks(project_id, title, done, priority) AS (
    VALUES
        (1, '调整配色', TRUE, 2),
        (1, '导出主题', FALSE, 3),
        (2, '检查图表', FALSE, 1)
)
-- 先关联并筛选，再按项目汇总未完成的任务。
SELECT
    p.name,
    COUNT(*) AS pending,
    SUM(CASE WHEN t.priority >= 3 THEN 1 ELSE 0 END) AS urgent,
    CASE WHEN COUNT(*) > 1 THEN '多项待办'
         ELSE '即将完成' END AS status
FROM projects AS p
JOIN tasks AS t ON t.project_id = p.id
WHERE t.done = FALSE
GROUP BY p.id, p.name
HAVING COUNT(*) > 0
ORDER BY urgent DESC, pending DESC;
```

```bash title="themes.sh · Bash" lineNumbers
#!/usr/bin/env bash
set -euo pipefail

root="${1:-cinderfolio/dist}"
files=("cf-rspress.css" "cf-pele.css" "cf-shiki-light.json")

report() {
    local path="$1"
    if [[ ! -f "$path" ]]; then
        printf '缺少文件：%s\n' "$path" >&2
        return 1
    fi
    case "$path" in
        *.json) printf '主题：%s\n' "${path##*/}" ;;
        *.css)  printf '样式：%s\n' "${path##*/}" ;;
    esac
}

# 函数的退出状态决定分支；缺少文件时继续检查下一项。
count=0
for file in "${files[@]}"; do
    if report "$root/$file"; then
        count=$((count + 1))
    else
        continue
    fi
done
printf '找到 %d / %d 个文件\n' "$count" "${#files[@]}"
printf '%s\n' "${files[@]}" | sort
```

```lua title="tasks.lua · Lua 5.4" lineNumbers
local tasks = {
    { id = 1, title = "调整配色", done = true },
    { id = 2, title = "导出主题", done = false },
}

local function pending(items)
    local result = {}
    for _, task in ipairs(items) do
        if not task.done then
            table.insert(result, task)
        end
    end
    table.sort(result, function(a, b) return a.id < b.id end)
    return result
end

local function describe(task)
    assert(task.title ~= "", "标题为空")
    return string.format("%d: %s", task.id, task.title)
end

-- pcall 将错误转换为状态值，单项失败不阻止后续处理。
for _, task in ipairs(pending(tasks)) do
    local ok, label = pcall(describe, task)
    if ok then
        print(label)
    else
        io.stderr:write(label, "\n")
    end
end

return { pending = pending, describe = describe }
```
